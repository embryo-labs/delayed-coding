cmake_minimum_required(VERSION 3.20)
project(DelayedCoding VERSION 0.1.0 LANGUAGES C CXX)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(DELAYED_CODING_RANS_DIR "" CACHE PATH "Optional pinned upstream ryg_rans checkout")
set(DELAYED_CODING_BLITZCRANK_DIR "" CACHE PATH "Optional original Blitzcrank checkout")
option(DELAYED_CODING_BUILD_TESTS "Build C ABI checks" ON)
option(DELAYED_CODING_REFERENCE_DIVISION "Use exact division instead of reciprocal (ablation)" OFF)
option(DELAYED_CODING_FLAT_ALIAS "Experimental flat alias layout (can regress model switching)" OFF)
option(DELAYED_CODING_SPECULATIVE_ENCODE "Experimental branchless multi-lane encoding" OFF)
option(DELAYED_CODING_BUILD_SIMD_BENCHMARK "Build separate SSE4.1 / 12-bit probability benchmark" OFF)
set(rust_target_dir "${CMAKE_CURRENT_BINARY_DIR}/cargo-target")
if(WIN32)
    set(rust_library "${rust_target_dir}/release/delayed_coding_ffi.lib")
else()
    set(rust_library "${rust_target_dir}/release/libdelayed_coding_ffi.a")
endif()
set(rust_features)
if(DELAYED_CODING_REFERENCE_DIVISION)
    list(APPEND rust_features --features delayed-coding/reference-division)
endif()
if(DELAYED_CODING_FLAT_ALIAS)
    list(APPEND rust_features --features delayed-coding/flat-alias)
endif()
if(DELAYED_CODING_SPECULATIVE_ENCODE)
    list(APPEND rust_features --features delayed-coding/speculative-encode)
endif()
file(GLOB_RECURSE rust_sources CONFIGURE_DEPENDS src/*.rs ffi/src/*.rs)
add_custom_command(OUTPUT "${rust_library}"
    COMMAND "${CARGO_EXECUTABLE}" build --release --offline --locked -p delayed-coding-ffi
        --target-dir "${rust_target_dir}" ${rust_features}
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS ${rust_sources} Cargo.toml Cargo.lock ffi/Cargo.toml
    VERBATIM)
add_custom_target(delayed_coding_rust ALL DEPENDS "${rust_library}")
add_library(delayed_coding STATIC IMPORTED GLOBAL)
add_library(DelayedCoding::delayed_coding ALIAS delayed_coding)
set_target_properties(delayed_coding PROPERTIES IMPORTED_LOCATION "${rust_library}"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_dependencies(delayed_coding delayed_coding_rust)
if(UNIX)
    find_package(Threads REQUIRED)
    set_property(TARGET delayed_coding PROPERTY INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS};m")
endif()
if(DELAYED_CODING_BUILD_TESTS)
    enable_testing()
    add_executable(c_roundtrip examples/c_roundtrip.c)
    target_link_libraries(c_roundtrip PRIVATE DelayedCoding::delayed_coding)
    add_test(NAME c_roundtrip COMMAND c_roundtrip)
endif()
if(DELAYED_CODING_BLITZCRANK_DIR)
    add_executable(legacy_branches tests/legacy_branches.cpp
        "${DELAYED_CODING_BLITZCRANK_DIR}/delayed_coding/src/utility.cpp")
    target_compile_features(legacy_branches PRIVATE cxx_std_17)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(legacy_branches PRIVATE -include cstdint)
    endif()
    target_include_directories(legacy_branches PRIVATE "${DELAYED_CODING_BLITZCRANK_DIR}/delayed_coding/include")
    target_link_libraries(legacy_branches PRIVATE DelayedCoding::delayed_coding)
    if(DELAYED_CODING_BUILD_TESTS)
        add_test(NAME legacy_branches COMMAND legacy_branches)
    endif()
    add_executable(legacy_differential tests/legacy_differential.cpp
        "${DELAYED_CODING_BLITZCRANK_DIR}/delayed_coding/src/utility.cpp")
    target_compile_features(legacy_differential PRIVATE cxx_std_17)
    # The research header relied on transitive <cstdint> includes removed in
    # newer libstdc++. Supply the missing standard header without editing oracle code.
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(legacy_differential PRIVATE -include cstdint)
    endif()
    target_include_directories(legacy_differential PRIVATE
        "${DELAYED_CODING_BLITZCRANK_DIR}/delayed_coding/include")
    target_link_libraries(legacy_differential PRIVATE DelayedCoding::delayed_coding)
    if(DELAYED_CODING_BUILD_TESTS)
        add_test(NAME legacy_differential COMMAND legacy_differential)
    endif()
endif()
if(DELAYED_CODING_RANS_DIR)
    if(NOT EXISTS "${DELAYED_CODING_RANS_DIR}/rans64.h")
        message(FATAL_ERROR "DELAYED_CODING_RANS_DIR must point to an upstream ryg_rans checkout")
    endif()
    set(upstream_loops "${CMAKE_CURRENT_BINARY_DIR}/upstream_loops.generated.h")
    add_custom_command(OUTPUT "${upstream_loops}"
        COMMAND "${CMAKE_COMMAND}" "-DUPSTREAM=${DELAYED_CODING_RANS_DIR}" "-DOUTPUT=${upstream_loops}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/extract_upstream.cmake"
        DEPENDS benchmarks/extract_upstream.cmake
            "${DELAYED_CODING_RANS_DIR}/rans_byte.h" "${DELAYED_CODING_RANS_DIR}/rans64.h"
            "${DELAYED_CODING_RANS_DIR}/rans_word_sse41.h" "${DELAYED_CODING_RANS_DIR}/platform.h"
            "${DELAYED_CODING_RANS_DIR}/main.cpp" "${DELAYED_CODING_RANS_DIR}/main64.cpp"
            "${DELAYED_CODING_RANS_DIR}/main_simd.cpp"
        VERBATIM)
    add_custom_target(generate_upstream_loops DEPENDS "${upstream_loops}")
    add_executable(compare_rans benchmarks/compare_rans.cpp)
    target_sources(compare_rans PRIVATE "${upstream_loops}")
    add_dependencies(compare_rans generate_upstream_loops)
    target_include_directories(compare_rans PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_features(compare_rans PRIVATE cxx_std_17)
    target_include_directories(compare_rans PRIVATE "${DELAYED_CODING_RANS_DIR}")
    target_link_libraries(compare_rans PRIVATE DelayedCoding::delayed_coding)
    # Upstream alias demo's platform.h uses x86 intrinsics and Linux/Windows timers.
    if((WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") AND
       CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$")
        target_compile_definitions(compare_rans PRIVATE DELAYED_CODING_HAVE_RANS_ALIAS=1)
    endif()
    if(DELAYED_CODING_BUILD_SIMD_BENCHMARK)
        if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$" OR
           NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR
           NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux"))
            message(FATAL_ERROR "The SIMD benchmark requires x86 Linux/Windows and GCC/Clang")
        endif()
        add_executable(compare_rans_simd benchmarks/compare_rans.cpp)
        target_sources(compare_rans_simd PRIVATE "${upstream_loops}")
        add_dependencies(compare_rans_simd generate_upstream_loops)
        target_include_directories(compare_rans_simd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_features(compare_rans_simd PRIVATE cxx_std_17)
        target_compile_options(compare_rans_simd PRIVATE -msse4.1)
        target_compile_definitions(compare_rans_simd PRIVATE
            DELAYED_CODING_HAVE_RANS_ALIAS=1 DELAYED_CODING_HAVE_RANS_SIMD=1)
        target_include_directories(compare_rans_simd PRIVATE "${DELAYED_CODING_RANS_DIR}")
        target_link_libraries(compare_rans_simd PRIVATE DelayedCoding::delayed_coding)
        add_executable(calibrate_upstream benchmarks/calibrate_upstream.cpp "${upstream_loops}")
        add_dependencies(calibrate_upstream generate_upstream_loops)
        target_compile_features(calibrate_upstream PRIVATE cxx_std_17)
        target_compile_options(calibrate_upstream PRIVATE -msse4.1 -UNDEBUG)
        target_compile_definitions(calibrate_upstream PRIVATE DELAYED_CODING_HAVE_RANS_SIMD=1)
        target_include_directories(calibrate_upstream PRIVATE "${DELAYED_CODING_RANS_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
        target_link_libraries(calibrate_upstream PRIVATE DelayedCoding::delayed_coding)
        add_executable(calibrate_upstream_scalar benchmarks/calibrate_upstream.cpp "${upstream_loops}")
        add_dependencies(calibrate_upstream_scalar generate_upstream_loops)
        target_compile_features(calibrate_upstream_scalar PRIVATE cxx_std_17)
        target_compile_options(calibrate_upstream_scalar PRIVATE -UNDEBUG)
        target_include_directories(calibrate_upstream_scalar PRIVATE "${DELAYED_CODING_RANS_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
        target_link_libraries(calibrate_upstream_scalar PRIVATE DelayedCoding::delayed_coding)
        if(DELAYED_CODING_BUILD_TESTS)
            add_test(NAME upstream_loop_calibration COMMAND calibrate_upstream --check)
            add_test(NAME upstream_scalar_calibration COMMAND calibrate_upstream_scalar --check)
        endif()
    endif()
    if(DELAYED_CODING_BUILD_TESTS)
        foreach(symbol_count 1 3 4 5 17 4097)
            add_test(NAME rans_roundtrip_${symbol_count}
                COMMAND compare_rans ${symbol_count} --check)
            if(DELAYED_CODING_BUILD_SIMD_BENCHMARK)
                add_test(NAME rans_simd_roundtrip_${symbol_count}
                    COMMAND compare_rans_simd ${symbol_count} --check)
            endif()
        endforeach()
        add_test(NAME rans_file_roundtrip
            COMMAND compare_rans --file "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" --check)
        add_test(NAME rans_records_roundtrip
            COMMAND compare_rans --records "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" --check)
        if(DELAYED_CODING_BUILD_SIMD_BENCHMARK)
            add_test(NAME rans_simd_file_roundtrip
                COMMAND compare_rans_simd --file "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" --check)
        endif()
    endif()
endif()
